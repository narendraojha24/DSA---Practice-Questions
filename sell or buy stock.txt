class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int length = prices.size();
         int buy = prices[0];
        int profit = 0;
        for (int i = 1; i < length; i++) {
            if (prices[i] < buy) {
                buy = prices[i];
            } else if (prices[i] - buy > profit) {
                profit = prices[i] - buy;
            }
        }
        return profit;
        
    }
};
